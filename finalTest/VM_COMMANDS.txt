# I run these commands from the Ubuntu VM after netmon starts on the Pi.
PI_IP=<pi-ip>

ping -c 1 "$PI_IP"
ssh -o ConnectTimeout=3 "$PI_IP"
printf 'netmon-dns-test\n' | nc -u -w 1 "$PI_IP" 53

for p in $(seq 1 30); do
  nc -zn -w 1 "$PI_IP" "$p" >/dev/null 2>&1 &
done
wait
